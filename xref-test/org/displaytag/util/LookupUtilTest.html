7   package org.displaytag.util;

9   import java.util.LinkedHashMap;
10  import java.util.List;
11  import java.util.Map;
12  
13  import org.displaytag.exception.ObjectLookupException;
14  import org.junit.jupiter.api.Assertions;
15  import org.junit.jupiter.api.Test;
16  import org.springframework.mock.web.MockPageContext;
17  
18  /**
19   * Tests for {@link LookupUtil}.
20   */
21  class LookupUtilTest {
22  
23      @Test
24      void testGetBeanValueSimpleAndNested() throws Exception {
25          final MockPageContext pageContext = new MockPageContext();
26          pageContext.setAttribute("simple", "value");
27          pageContext.setAttribute("bean", new ParentBean(new ChildBean("nested")));
28  
29          Assertions.assertEquals("value", LookupUtil.getBeanValue(pageContext, "simple", MockPageContext.PAGE_SCOPE));
30          Assertions.assertEquals("nested",
31                  LookupUtil.getBeanValue(pageContext, "bean.child.value", MockPageContext.PAGE_SCOPE));
32          Assertions.assertNull(LookupUtil.getBeanValue(pageContext, "missing.property", MockPageContext.PAGE_SCOPE));
33      }
34  
35      @Test
36      void testGetBeanPropertyAndGetPropertyForMapAndIndexedValues() throws Exception {
37          final ParentBean bean = new ParentBean(new ChildBean("x"));
38          bean.setList(List.of("zero", "one", "two"));
39  
40          Assertions.assertEquals("x", LookupUtil.getBeanProperty(bean, "child.value"));
41          Assertions.assertEquals("one", LookupUtil.getProperty(bean, "list[1]"));
42          Assertions.assertEquals("one", LookupUtil.getIndexedProperty(List.of("zero", "one", "two"), "[1]"));
43          Assertions.assertEquals("two", LookupUtil.getIndexedProperty(new String[] { "zero", "one", "two" }, "[2]"));
44          Assertions.assertEquals("mapped", LookupUtil.getProperty(Map.of("k", "mapped"), "k"));
45          Assertions.assertEquals("mappedValue", LookupUtil.getProperty(new MappedBean(), "mapped(m)"));
46          final Map<String, Object> nullMap = new LinkedHashMap<>();
47          nullMap.put("child", null);
48          Assertions.assertNull(LookupUtil.getProperty(nullMap, "child.value"));
49      }
50  
51      @Test
52      void testLookupFailuresAndNullChecks() {
53          Assertions.assertThrows(NullPointerException.class, () -> LookupUtil.getBeanProperty(null, "x"));
54          Assertions.assertThrows(NullPointerException.class, () -> LookupUtil.getBeanProperty(new Object(), null));
55          Assertions.assertThrows(IllegalArgumentException.class, () -> LookupUtil.getIndexedProperty(List.of("a"), "x"));
56          Assertions.assertThrows(IllegalArgumentException.class,
57                  () -> LookupUtil.getIndexedProperty(List.of("a"), "[x]"));
58          Assertions.assertThrows(ObjectLookupException.class,
59                  () -> LookupUtil.getBeanProperty(new ParentBean(new ChildBean("x")), "missing"));
60      }
61  
62      public static class ParentBean {
63          private final ChildBean child;
64          private List<String> list;
65  
66          ParentBean(final ChildBean child) {
67              this.child = child;
68          }
69  
70          public ChildBean getChild() {
71              return this.child;
72          }
73  
74          public List<String> getList() {
75              return this.list;
76          }
77  
78          public void setList(final List<String> list) {
79              this.list = list;
80          }
81      }
82  
83      public static class MappedBean {
84          public String getMapped(final String key) {
85              return "m".equals(key) ? "mappedValue" : null;
86          }
87      }
88  
89      public static class ChildBean {
90          private final String value;
91  
92          ChildBean(final String value) {
93              this.value = value;
94          }
95  
96          public String getValue() {
97              return this.value;
98          }
99      }
100 }
