7   package org.displaytag.filter;
8   
9   import java.io.PrintWriter;
10  
11  import org.displaytag.tags.TableTagParameters;
12  import org.junit.jupiter.api.Assertions;
13  import org.junit.jupiter.api.Test;
14  import org.springframework.mock.web.MockHttpServletResponse;
15  
16  /**
17   * Tests for {@link BufferedResponseWrapper13Impl}.
18   */
19  class BufferedResponseWrapper13ImplTest {
20  
21      @Test
22      void testContentTypeEncodingAndBufferedWriters() throws Exception {
23          final MockHttpServletResponse response = new MockHttpServletResponse();
24          final BufferedResponseWrapper13Impl wrapper = new BufferedResponseWrapper13Impl(response);
25  
26          wrapper.setContentType("text/plain; charset=UTF-8");
27          wrapper.setCharacterEncoding("UTF-16");
28          Assertions.assertEquals("text/plain", wrapper.getContentType());
29          Assertions.assertEquals("UTF-16", wrapper.getCharacterEncoding());
30  
31          final PrintWriter writer = wrapper.getWriter();
32          writer.write("hello");
33          writer.flush();
34          wrapper.getOutputStream().write('!');
35          wrapper.flushBuffer();
36          Assertions.assertTrue(wrapper.getContentAsString().contains("hello"));
37      }
38  
39      @Test
40      void testStateSwitchAndHeaderFiltering() throws Exception {
41          final MockHttpServletResponse response = new MockHttpServletResponse();
42          final BufferedResponseWrapper13Impl wrapper = new BufferedResponseWrapper13Impl(response);
43  
44          wrapper.addHeader("x-header", "value");
45          wrapper.addHeader("cache-control", "no-cache");
46          Assertions.assertEquals("value", response.getHeader("x-header"));
47          Assertions.assertNull(response.getHeader("cache-control"));
48  
49          wrapper.setDateHeader("x-date", 1L);
50          wrapper.addDateHeader("x-date2", 2L);
51          wrapper.setIntHeader("x-int", 3);
52          wrapper.addIntHeader("x-int2", 4);
53          wrapper.setHeader("x-set", "v");
54          Assertions.assertEquals("v", response.getHeader("x-set"));
55  
56          wrapper.addHeader(TableTagParameters.PARAMETER_EXPORTING, "true");
57          wrapper.setContentType("application/pdf");
58          wrapper.setCharacterEncoding("UTF-8");
59          wrapper.getOutputStream();
60          Assertions.assertTrue(wrapper.isOutRequested());
61      }
62  }
