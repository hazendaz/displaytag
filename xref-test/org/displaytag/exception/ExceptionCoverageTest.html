7   package org.displaytag.exception;
8   
9   import org.displaytag.decorator.DefaultDecoratorFactory;
10  import org.displaytag.model.RowSorter;
11  import org.displaytag.properties.TableProperties;
12  import org.junit.jupiter.api.Assertions;
13  import org.junit.jupiter.api.Test;
14  
15  /**
16   * Additional coverage tests for exception classes.
17   */
18  class ExceptionCoverageTest {
19  
20      @Test
21      void testBaseNestableRuntimeExceptionSeverityBranchesAndCause() {
22          final RuntimeException cause = new RuntimeException("cause");
23  
24          final TestRuntimeException debug = new TestRuntimeException(ExceptionCoverageTest.class, "debug",
25                  SeverityEnum.DEBUG);
26          final TestRuntimeException info = new TestRuntimeException(ExceptionCoverageTest.class, "info",
27                  SeverityEnum.INFO);
28          final TestRuntimeException warn = new TestRuntimeException(ExceptionCoverageTest.class, "warn",
29                  SeverityEnum.WARN);
30          final TestRuntimeException error = new TestRuntimeException(ExceptionCoverageTest.class, "error", cause,
31                  SeverityEnum.ERROR);
32  
33          Assertions.assertNull(debug.getCause());
34          Assertions.assertNull(info.getCause());
35          Assertions.assertNull(warn.getCause());
36          Assertions.assertSame(cause, error.getCause());
37          Assertions.assertTrue(error.toString().contains("error"));
38          Assertions.assertTrue(error.toString().contains("cause"));
39      }
40  
41      @Test
42      void testJspAndRuntimeConcreteExceptions() {
43          final Throwable cause = new IllegalArgumentException("boom");
44  
45          final DecoratorException decoratorException = new DecoratorException(ExceptionCoverageTest.class, "decorator");
46          final DecoratorInstantiationException instantiationException = new DecoratorInstantiationException(
47                  DefaultDecoratorFactory.class, "decorator", cause);
48          final MissingAttributeException missingAttributeException = new MissingAttributeException(
49                  ExceptionCoverageTest.class, new String[] { "a", "b" });
50          final ObjectLookupException lookupException = new ObjectLookupException(ExceptionCoverageTest.class, this, "x",
51                  cause);
52          final RuntimeLookupException runtimeLookupException = new RuntimeLookupException(RowSorter.class, "x",
53                  lookupException);
54          final TagStructureException tagStructureException = new TagStructureException(ExceptionCoverageTest.class,
55                  "column", "table");
56          final WrappedRuntimeException wrappedRuntimeException = new WrappedRuntimeException(ExceptionCoverageTest.class,
57                  cause);
58          final TablePropertiesLoadException tablePropertiesLoadException = new TablePropertiesLoadException(
59                  ExceptionCoverageTest.class, "displaytag.properties", cause);
60          final FactoryInstantiationException factoryInstantiationException = new FactoryInstantiationException(
61                  TableProperties.class, "factory.decorator", "x.Factory", cause);
62  
63          Assertions.assertEquals(SeverityEnum.ERROR, decoratorException.getSeverity());
64          Assertions.assertEquals(SeverityEnum.ERROR, instantiationException.getSeverity());
65          Assertions.assertEquals(SeverityEnum.ERROR, missingAttributeException.getSeverity());
66          Assertions.assertEquals(SeverityEnum.WARN, lookupException.getSeverity());
67          Assertions.assertEquals(SeverityEnum.ERROR, tagStructureException.getSeverity());
68          Assertions.assertEquals(SeverityEnum.WARN, wrappedRuntimeException.getSeverity());
69          Assertions.assertEquals(SeverityEnum.ERROR, tablePropertiesLoadException.getSeverity());
70          Assertions.assertEquals(SeverityEnum.FATAL, factoryInstantiationException.getSeverity());
71          Assertions.assertArrayEquals(new String[] { "a", "b" }, missingAttributeException.getAttributeNames());
72          Assertions.assertTrue(runtimeLookupException.getMessage().contains("x"));
73          Assertions.assertEquals("debug", SeverityEnum.DEBUG.getSeverity());
74      }
75  
76      private static final class TestRuntimeException extends BaseNestableRuntimeException {
77          private static final long serialVersionUID = 1L;
78          private final SeverityEnum severity;
79  
80          private TestRuntimeException(final Class<?> source, final String message, final SeverityEnum severity) {
81              super(source, message);
82              this.severity = severity;
83          }
84  
85          private TestRuntimeException(final Class<?> source, final String message, final Throwable cause,
86                  final SeverityEnum severity) {
87              super(source, message, cause);
88              this.severity = severity;
89          }
90  
91          @Override
92          public SeverityEnum getSeverity() {
93              return this.severity;
94          }
95      }
96  }
