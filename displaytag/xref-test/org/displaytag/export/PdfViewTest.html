7   package org.displaytag.export;
8   
9   import jakarta.servlet.jsp.JspException;
10  
11  import java.io.ByteArrayOutputStream;
12  
13  import org.displaytag.exception.SeverityEnum;
14  import org.displaytag.model.HeaderCell;
15  import org.displaytag.model.Row;
16  import org.displaytag.model.TableModel;
17  import org.displaytag.properties.TableProperties;
18  import org.displaytag.util.HtmlAttributeMap;
19  import org.junit.jupiter.api.Assertions;
20  import org.junit.jupiter.api.Test;
21  
22  class PdfViewTest {
23  
24      @Test
25      void testDoExportGeneratesPdf() throws Exception {
26          final PdfView view = new PdfView();
27          final TableModel model = createModel();
28          view.setParameters(model, true, true, true);
29  
30          final ByteArrayOutputStream out = new ByteArrayOutputStream();
31          view.doExport(out);
32  
33          Assertions.assertEquals("application/pdf", view.getMimeType());
34          Assertions.assertTrue(out.size() > 0);
35      }
36  
37      @Test
38      void testDoExportWrapsFailures() {
39          final PdfView view = new PdfView() {
40              @Override
41              protected void generatePDFTable() throws JspException {
42                  throw new JspException("boom");
43              }
44          };
45  
46          view.setParameters(createModel(), true, true, true);
47  
48          final PdfView.PdfGenerationException ex = Assertions.assertThrows(PdfView.PdfGenerationException.class,
49                  () -> view.doExport(new ByteArrayOutputStream()));
50          Assertions.assertEquals(SeverityEnum.ERROR, ex.getSeverity());
51      }
52  
53      private static TableModel createModel() {
54          final TableProperties properties = TableProperties.getInstance(null);
55          final TableModel model = new TableModel(properties, "pdf", null);
56          model.setRowListPage(model.getRowListFull());
57  
58          final HeaderCell first = new HeaderCell();
59          first.setTitle("Visible title");
60          first.setBeanPropertyName("first");
61          first.setHtmlAttributes(new HtmlAttributeMap());
62          model.addColumnHeader(first);
63  
64          final HeaderCell second = new HeaderCell();
65          second.setTitle(null);
66          second.setBeanPropertyName("second");
67          second.setHtmlAttributes(new HtmlAttributeMap());
68          model.addColumnHeader(second);
69  
70          model.addRow(new Row(new PdfBean("left", "right"), 0));
71          return model;
72      }
73  
74      public static class PdfBean {
75          private final String first;
76          private final String second;
77  
78          PdfBean(final String first, final String second) {
79              this.first = first;
80              this.second = second;
81          }
82  
83          public String getFirst() {
84              return this.first;
85          }
86  
87          public String getSecond() {
88              return this.second;
89          }
90      }
91  }
