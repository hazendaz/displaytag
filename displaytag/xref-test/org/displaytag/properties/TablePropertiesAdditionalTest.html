25      @Test
26      void testDefaultFallbackGettersAndBooleanBranches() throws Exception {
27          TableProperties.clearProperties();
28          TableProperties.setLocaleResolver(null);
29          TableProperties.setResourceProvider(null);
30          final TableProperties properties = TableProperties.getInstance(null);
31  
32          removeProperty(properties, TableProperties.PROPERTY_INT_PAGING_GROUPSIZE);
33          removeProperty(properties, TableProperties.PROPERTY_STRING_PAGINATION_SORT_PARAM);
34          removeProperty(properties, TableProperties.PROPERTY_STRING_PAGINATION_PAGE_NUMBER_PARAM);
35          removeProperty(properties, TableProperties.PROPERTY_STRING_PAGINATION_SORT_DIRECTION_PARAM);
36          removeProperty(properties, TableProperties.PROPERTY_STRING_PAGINATION_SEARCH_ID_PARAM);
37          removeProperty(properties, TableProperties.PROPERTY_STRING_PAGINATION_ASC_VALUE);
38          removeProperty(properties, TableProperties.PROPERTY_STRING_PAGINATION_DESC_VALUE);
39          removeProperty(properties, TableProperties.PROPERTY_BOOLEAN_PAGINATION_SKIP_PAGE_NUMBER_IN_SORT);
40          removeProperty(properties, TableProperties.PROPERTY_BOOLEAN_LEGACY_FORM_SUBMIT);
41  
42          Assertions.assertEquals(8, properties.getPagingGroupSize());
43          Assertions.assertEquals("sort", properties.getPaginationSortParam());
44          Assertions.assertEquals("page", properties.getPaginationPageNumberParam());
45          Assertions.assertEquals("dir", properties.getPaginationSortDirectionParam());
46          Assertions.assertEquals("searchid", properties.getPaginationSearchIdParam());
47          Assertions.assertEquals("asc", properties.getPaginationAscValue());
48          Assertions.assertEquals("desc", properties.getPaginationDescValue());
49          Assertions.assertTrue(properties.getPaginationSkipPageNumberInSort());
50          Assertions.assertFalse(properties.getUseLegacyFormSubmit());
51          Assertions.assertTrue(properties.getExportTypes().length > 0);
52  
53          properties.setProperty(TableProperties.PROPERTY_STRING_BANNER_PLACEMENT, "top");
54          Assertions.assertTrue(properties.getAddPagingBannerTop());
55          Assertions.assertFalse(properties.getAddPagingBannerBottom());
56  
57          properties.setProperty(TableProperties.PROPERTY_STRING_BANNER_PLACEMENT, "bottom");
58          Assertions.assertFalse(properties.getAddPagingBannerTop());
59          Assertions.assertTrue(properties.getAddPagingBannerBottom());
60  
61          properties.setProperty(TableProperties.PROPERTY_STRING_EXPORTBANNER_PLACEMENT, "top");
62          Assertions.assertTrue(properties.getAddExportBannerTop());
63          Assertions.assertFalse(properties.getAddExportBannerBottom());
64  
65          properties.setProperty(TableProperties.PROPERTY_STRING_EXPORTBANNER_PLACEMENT, "both");
66          Assertions.assertTrue(properties.getAddExportBannerTop());
67          Assertions.assertTrue(properties.getAddExportBannerBottom());
68      }

70      @Test
71      void testFactoryAndComparatorLoadingBranches() {
72          TableProperties.clearProperties();
73          TableProperties.setLocaleResolver(null);
74          TableProperties.setResourceProvider(null);
75          final TableProperties properties = TableProperties.getInstance(null);
76  
77          properties.setProperty(TableProperties.PROPERTY_CLASS_REQUESTHELPERFACTORY, InvalidClass.class.getName());
78          Assertions.assertThrows(FactoryInstantiationException.class, properties::getRequestHelperFactoryInstance);
79  
80          properties.setProperty(TableProperties.PROPERTY_CLASS_REQUESTHELPERFACTORY,
81                  TestRequestHelperFactory.class.getName());
82          final RequestHelperFactory helperFactory = properties.getRequestHelperFactoryInstance();
83          Assertions.assertTrue(helperFactory instanceof TestRequestHelperFactory);
84  
85          properties.setProperty(TableProperties.PROPERTY_CLASS_DECORATORFACTORY, TestDecoratorFactory.class.getName());
86          final DecoratorFactory decoratorFactory = properties.getDecoratorFactoryInstance();
87          Assertions.assertTrue(decoratorFactory instanceof TestDecoratorFactory);
88  
89          properties.setProperty(TableProperties.PROPERTY_DEFAULT_COMPARATOR, TestComparator.class.getName());
90          final Comparator<Object> comparator = properties.getDefaultComparator();
91          Assertions.assertTrue(comparator instanceof TestComparator);
92  
93          properties.setProperty(TableProperties.PROPERTY_CLASS_LOCALEPROVIDER, TestResourceProvider.class.getName());
94          final I18nResourceProvider provider = properties.geResourceProvider();
95          Assertions.assertTrue(provider instanceof TestResourceProvider);
96      }

100         TableProperties.clearProperties();
101         TableProperties.setLocaleResolver(null);

116     @Test
117     void testInvalidClassAndInvalidNumericConfigurationBranches() {
118         TableProperties.clearProperties();
119         TableProperties.setLocaleResolver(null);
120         TableProperties.setResourceProvider(null);
121         final TableProperties properties = TableProperties.getInstance(null);
122 
123         properties.setProperty(TableProperties.PROPERTY_CLASS_DECORATORFACTORY, String.class.getName());
124         Assertions.assertThrows(FactoryInstantiationException.class, properties::getDecoratorFactoryInstance);
125 
126         properties.setProperty(TableProperties.PROPERTY_DEFAULT_COMPARATOR, "missing.Comparator");
127         Assertions.assertFalse(properties.getDefaultComparator() instanceof TestComparator);
128 
129         properties.setProperty(TableProperties.PROPERTY_INT_PAGING_GROUPSIZE, "invalid");
130         Assertions.assertEquals(8, properties.getPagingGroupSize());
131     }

133     private static void removeProperty(final TableProperties properties, final String key) throws Exception {
134         final Field field = TableProperties.class.getDeclaredField("properties");
135         field.setAccessible(true);
136         final Properties raw = (Properties) field.get(properties);
137         raw.remove(key);
138     }
