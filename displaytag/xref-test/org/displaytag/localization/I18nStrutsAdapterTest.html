26  class I18nStrutsAdapterTest {
27  
28      @Test
29      void testResolveLocaleFromSessionAndRequest() {
30          final I18nStrutsAdapter adapter = new I18nStrutsAdapter();
31          final MockHttpServletRequest request = new MockHttpServletRequest();
32          request.addPreferredLocale(Locale.CANADA_FRENCH);
33          final MockPageContext pageContext = new MockPageContext(new MockServletContext(), request);
34  
35          Assertions.assertEquals(Locale.CANADA_FRENCH, adapter.resolveLocale(pageContext));
36  
37          request.getSession(true).setAttribute(Globals.LOCALE_KEY, Locale.ITALIAN);
38          Assertions.assertEquals(Locale.ITALIAN, adapter.resolveLocale(pageContext));
39      }
40  
41      @Test
42      void testGetResourceLookupPathsAndUndefinedFallback() {
43          final I18nStrutsAdapter adapter = new I18nStrutsAdapter();
44          final MockHttpServletRequest request = new MockHttpServletRequest();
45          request.addPreferredLocale(Locale.US);
46          final MockPageContext pageContext = new MockPageContext(new MockServletContext(), request);
47          final TestMessageResources requestResources = new TestMessageResources(Map.of("title", "request-title"));
48  
49          pageContext.setAttribute(Globals.MESSAGES_KEY, requestResources, PageContext.REQUEST_SCOPE);
50          Assertions.assertEquals("request-title", adapter.getResource("title", "ignored", null, pageContext));
51  
52          pageContext.removeAttribute(Globals.MESSAGES_KEY, PageContext.REQUEST_SCOPE);
53          final ModuleConfig moduleConfig = (ModuleConfig) Proxy.newProxyInstance(ModuleConfig.class.getClassLoader(),
54                  new Class<?>[] { ModuleConfig.class }, new ModuleConfigHandler("/module"));
55          request.setAttribute(Globals.MODULE_KEY, moduleConfig);
56          pageContext.setAttribute(Globals.MESSAGES_KEY + "/module",
57                  new TestMessageResources(Map.of("title", "module-title")), PageContext.APPLICATION_SCOPE);
58          Assertions.assertEquals("module-title", adapter.getResource("title", "ignored", null, pageContext));
59  
60          pageContext.removeAttribute(Globals.MESSAGES_KEY + "/module", PageContext.APPLICATION_SCOPE);
61          Assertions.assertEquals(I18nStrutsAdapter.UNDEFINED_KEY + "missing" + I18nStrutsAdapter.UNDEFINED_KEY,
62                  adapter.getResource("missing", "ignored", null, pageContext));
63          Assertions.assertNull(adapter.getResource(null, "missingDefault", null, pageContext));
64      }
65  
66      private static class TestMessageResources extends MessageResources {
67          private static final long serialVersionUID = 1L;
68  
69          private final Map<String, String> messages = new HashMap<>();
70  
71          TestMessageResources(final Map<String, String> messages) {
72              super(null, "test");
73              this.messages.putAll(messages);
74          }
75  
76          @Override
77          public String getMessage(final Locale locale, final String key) {
78              return this.messages.get(key);
79          }
80      }
81  
82      private static class ModuleConfigHandler implements InvocationHandler {
83          private final String prefix;
84  
85          ModuleConfigHandler(final String prefix) {
86              this.prefix = prefix;
87          }
88  
89          @Override
90          public Object invoke(final Object proxy, final java.lang.reflect.Method method, final Object[] args) {
91              if ("getPrefix".equals(method.getName())) {
92                  return this.prefix;
93              }
94              if ("setPrefix".equals(method.getName())) {
95                  return null;
96              }
97              if ("getConfigured".equals(method.getName())) {
98                  return Boolean.TRUE;
99              }
100             if ("equals".equals(method.getName())) {
101                 return proxy == args[0];
102             }
103             if ("hashCode".equals(method.getName())) {
104                 return System.identityHashCode(proxy);
105             }
106             return null;
107         }
108     }
